#!/usr/bin/env bash set -euo pipefail REPO_DIR="/var/www/houseofcauliman.com" FEED_JSON="$REPO_DIR/assets/monolith/feed.json" # Locked wallets MONOLITH_WALLET="r9iKXkMGF2HxAEsCLqdrghWpJmfFZN1Gw8" CREATOR_WALLET="rfgFM8z2abu2uahXFaRre8aWw9PEv48QbS" die(){ echo "[monolith_feed_add] ERROR: $*" >&2; exit 1; } need_bin(){ command -v "$1" >/dev/null 2>&1 || die "missing dependency: $1"; } need_bin jq need_bin git if [[ ! -d "$REPO_DIR/.git" ]]; then die "repo not found at $REPO_DIR" fi cd "$REPO_DIR" if [[ ! -f "$FEED_JSON" ]]; then die "feed.json not found at $FEED_JSON" fi mask_wallet(){ local w="$1" local a="${w:0:4}" local b="${w: -3}" echo "${a}...${b}" } tile_from_coords(){ local gx="$1" local gy="$2" local ax ay sx sy ax="${gx#-}"; ay="${gy#-}" if [[ "$gx" =~ ^- ]]; then sx="W"; else sx="E"; fi if [[ "$gy" -le 0 ]]; then sy="N"; ay="${ay#-}"; else sy="S"; fi echo "${sx}${ax}-${sy}${ay}" } valid_tile(){ [[ "$1" =~ ^[EW][0-9]+-[NS][0-9]+$ ]] } valid_tag(){ local t="$1" [[ -z "$t" ]] && return 0 [[ "${#t}" -le 24 ]] || return 1 [[ "$t" =~ [\<\>\ $'\n'$'\r'] ]] && return 1 return 0 } valid_wallet(){ local w="$1" [[ "$w" =~ ^r[1-9A-HJ-NP-Za-km-z]{24,34}$ ]] } valid_x_handle(){ local v="$1" [[ "$v" =~ ^@[A-Za-z0-9_]{1,15}$ ]] } valid_https_url(){ local v="$1" [[ "$v" == https://* ]] || return 1 [[ "${#v}" -ge 9 && "${#v}" -le 120 ]] || return 1 [[ "$v" =~ [\<\>\ ] ]] && return 1 return 0 } valid_link(){ local v="$1" [[ -z "$v" ]] && return 0 valid_x_handle "$v" && return 0 valid_https_url "$v" && return 0 return 1 } valid_note(){ local n="$1" [[ -z "$n" ]] && return 0 [[ "${#n}" -le 140 ]] || return 1 [[ "$n" =~ [\<\>] ]] && return 1 return 0 } normalize_link(){ local v="$1" v="$(echo -n "$v" | tr -d '[:space:]')" echo -n "$v" } today_iso(){ date -u +"%Y-%m-%d" } ensure_feed_array(){ jq -e 'type=="array"' "$FEED_JSON" >/dev/null 2>&1 || die "feed.json must be a JSON array" } ensure_special_tiles(){ local monolith_tile="E0-N0" local creator_tile="E1-N0" ensure_feed_array local tmp tmp="$(mktemp)" jq --arg mt "$monolith_tile" \ --arg mw "$MONOLITH_WALLET" \ --arg mtag "MONOLITH" \ --arg mlink "@monolith_wall" \ --arg ct "$creator_tile" \ --arg ctag "MRCAULIMAN" \ --arg cw "$CREATOR_WALLET" \ --arg clink "@mrcauliman" \ ' def upsert($tile; $obj): (map(select(.tile != $tile)) + [$obj]); . as $arr | ($arr | map(select(.tile != $mt and .tile != $ct))) as $rest | $rest | upsert($mt; { tile:$mt, gx:0, gy:0, tag:$mtag, wallet:$mw, wallet_public:false, link:$mlink, note:"Leave your mark. Live forever.", ts:(.[]? | select(.tile==$mt) | .ts) // (now|strftime("%Y-%m-%d")) }) | upsert($ct; { tile:$ct, gx:1, gy:0, tag:$ctag, wallet:$cw, wallet_public:false, link:$clink, note:"Foundational wallet. First liquidity. First fuel.", ts:(.[]? | select(.tile==$ct) | .ts) // (now|strftime("%Y-%m-%d")) }) ' "$FEED_JSON" > "$tmp" mv "$tmp" "$FEED_JSON" } usage(){ cat <<'EOF' monolith_feed_add adds or updates a tile entry in assets/monolith/feed.json Usage monolith_feed_add /path/to/claim.json claim.json must match the claim page output, example: { "tile":"E23-N11", "gx":23, "gy":-11, "tag":"YOURTAG", "wallet":"r....", "wallet_public":false, "link":"@username or https://...", "note":"legacy line", "ts":"2026-02-06" } Rules enforced tile must be E/W + number + - + N/S + number wallet required and must be XRPL r-address tag optional max 24, no <> or control chars link optional, single value, @handle or https URL only note optional max 140, no <> wallet_public default false if missing Special tiles enforced every run E0-N0 -> MONOLITH + hidden wallet + @monolith_wall E1-N0 -> MRCAULIMAN + hidden wallet + @mrcauliman EOF } [[ "${1:-}" == "-h" || "${1:-}" == "--help" ]] && { usage; exit 0; } [[ -z "${1:-}" ]] && { usage; exit 1; } CLAIM_PATH="$1" [[ -f "$CLAIM_PATH" ]] || die "claim file not found: $CLAIM_PATH" ensure_feed_array # read fields tile="$(jq -r '.tile // empty' "$CLAIM_PATH")" gx="$(jq -r '.gx // empty' "$CLAIM_PATH")" gy="$(jq -r '.gy // empty' "$CLAIM_PATH")" tag="$(jq -r '.tag // empty' "$CLAIM_PATH")" wallet="$(jq -r '.wallet // empty' "$CLAIM_PATH")" wallet_public="$(jq -r '.wallet_public // false' "$CLAIM_PATH")" link="$(jq -r '.link // empty' "$CLAIM_PATH")" note="$(jq -r '.note // empty' "$CLAIM_PATH")" ts="$(jq -r '.ts // empty' "$CLAIM_PATH")" [[ -n "$tile" ]] || die "tile missing" tile="$(echo -n "$tile" | tr '[:lower:]' '[:upper:]' | tr -d '[:space:]' | sed 's/[–—−]/-/g')" if [[ -n "$gx" && -n "$gy" && ( "$tile" == "null" || "$tile" == "" ) ]]; then tile="$(tile_from_coords "$gx" "$gy")" fi valid_tile "$tile" || die "tile invalid: $tile" # coords must align with tile, but we tolerate missing gx/gy and recompute if [[ -z "$gx" || -z "$gy" || "$gx" == "null" || "$gy" == "null" ]]; then if [[ "$tile" =~ ^([EW])([0-9]+)-([NS])([0-9]+)$ ]]; then sx="${BASH_REMATCH[1]}"; ax="${BASH_REMATCH[2]}" sy="${BASH_REMATCH[3]}"; ay="${BASH_REMATCH[4]}" if [[ "$sx" == "W" ]]; then gx="-$ax"; else gx="$ax"; fi if [[ "$sy" == "N" ]]; then gy="-$ay"; else gy="$ay"; fi fi fi [[ "$gx" =~ ^-?[0-9]+$ ]] || die "gx invalid" [[ "$gy" =~ ^-?[0-9]+$ ]] || die "gy invalid" # normalize tag tag="$(echo -n "$tag" | tr -d '\r' | sed 's/^[[:space:]]\+//; s/[[:space:]]\+$//')" [[ "$tag" == "null" ]] && tag="" if [[ -n "$tag" ]]; then valid_tag "$tag" || die "tag invalid" fi # wallet wallet="$(echo -n "$wallet" | tr -d '\r' | sed 's/^[[:space:]]\+//; s/[[:space:]]\+$//')" [[ -n "$wallet" ]] || die "wallet required" valid_wallet "$wallet" || die "wallet invalid" # wallet_public normalize if [[ "$wallet_public" == "true" || "$wallet_public" == "false" ]]; then : else wallet_public="false" fi # link normalize link="$(normalize_link "$link")" [[ "$link" == "null" ]] && link="" valid_link "$link" || die "link invalid" # note note="$(echo -n "$note" | tr -d '\r' | sed 's/^[[:space:]]\+//; s/[[:space:]]\+$//')" [[ "$note" == "null" ]] && note="" valid_note "$note" || die "note invalid" # ts default today ts="$(echo -n "$ts" | tr -d '\r' | sed 's/^[[:space:]]\+//; s/[[:space:]]\+$//')" [[ "$ts" == "null" ]] && ts="" if [[ -z "$ts" ]]; then ts="$(today_iso)" fi [[ "$ts" =~ ^[0-9]{4}-[0-9]{2}-[0-9]{2}$ ]] || die "ts invalid, expected YYYY-MM-DD" # upsert into feed tmp="$(mktemp)" jq --arg tile "$tile" \ --argjson gx "$gx" \ --argjson gy "$gy" \ --arg tag "$tag" \ --arg wallet "$wallet" \ --argjson wallet_public "$wallet_public" \ --arg link "$link" \ --arg note "$note" \ --arg ts "$ts" \ ' def clean_nulls: with_entries(select(.value != "" and .value != null)); . as $arr | (map(select(.tile != $tile))) + [( { tile:$tile, gx:$gx, gy:$gy, tag: ($tag | if .=="" then null else . end), wallet:$wallet, wallet_public:$wallet_public, link: ($link | if .=="" then null else . end), note: ($note | if .=="" then null else . end), ts:$ts } | clean_nulls )] ' "$FEED_JSON" > "$tmp" mv "$tmp" "$FEED_JSON" # enforce special tiles every time ensure_special_tiles # format feed as pretty JSON tmp2="$(mktemp)" jq '.' "$FEED_JSON" > "$tmp2" mv "$tmp2" "$FEED_JSON" echo "[monolith_feed_add] updated $FEED_JSON" echo "[monolith_feed_add] tile $tile wallet $(mask_wallet "$wallet") public=$wallet_public link=${link:-None}"
